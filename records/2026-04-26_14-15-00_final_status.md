```python
# 检查 transformers 版本
import transformers
tv = transformers.__version__.split('.')
major, minor = int(tv[0]), int(tv[1])
if major < 4 or (major == 4 and minor < 57):
    raise HTTPException(
        status_code=503,
        detail=f"Qwen3-TTS 需要 transformers >= 4.57.0..."
    )
```
