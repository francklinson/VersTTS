## 问题描述

在离线部署测试中，发现 IndexTTS2 仍然尝试访问 HuggingFace 下载模型，导致以下错误：

```
MaxRetryError: HTTPSConnectionPool(host='huggingface.co', port=443): 
Max retries exceeded with url: /facebook/w2v-bert-2.0/resolve/main/preprocessor_config.json
```

涉及的模型包括：
1. `facebook/w2v-bert-2.0` - SeamlessM4TFeatureExtractor
2. `amphion/MaskGCT` - semantic_codec
3. `funasr/campplus` - campplus 模型
4. `nvidia/bigvgan_v2_22khz_80band_256x` - BigVGAN

### 1. 修改 `algorithms/IndexTTS/indextts/infer_v2.py`

#### 1.1 添加离线模式检测函数

```python
def is_offline_mode():
    """检查是否处于离线模式"""
    return os.environ.get('TRANSFORMERS_OFFLINE') == '1' or os.environ.get('HF_HUB_OFFLINE') == '1'


def get_local_model_path(model_name, local_dir=None):
    """
    获取本地模型路径
    Args:
        model_name: HuggingFace 模型名称
        local_dir: 本地模型目录
    Returns:
        本地模型路径或 None
    """
    if local_dir and os.path.exists(local_dir):
        return local_dir
    
    # 检查 HF_HOME 缓存
    hf_home = os.environ.get('HF_HOME', os.path.expanduser('~/.cache/huggingface'))
    model_cache_dir = os.path.join(hf_home, 'hub', f'models--{model_name.replace("/", "--")}')
    
    if os.path.exists(model_cache_dir):
        # 查找快照目录
        for root, dirs, files in os.walk(model_cache_dir):
            if 'config.json' in files:
                return root
    
    return None
```

    print(">> Loading w2v-bert-2.0 from HuggingFace...")

# 加载 BigVGAN，离线模式下使用本地模型
bigvgan_name = self.cfg.vocoder.name

if os.path.exists(bigvgan_local):
    print(f">> Loading BigVGAN from local path: {bigvgan_local}")
    self.bigvgan = bigvgan.BigVGAN.from_pretrained(
        bigvgan_local, 
        use_cuda_kernel=self.use_cuda_kernel,
        local_files_only=True
    )
elif is_offline_mode():
    # 检查是否是本地路径
    if os.path.isdir(bigvgan_name):
        print(f">> Loading BigVGAN from local path: {bigvgan_name}")
        self.bigvgan = bigvgan.BigVGAN.from_pretrained(
            bigvgan_name, 
            use_cuda_kernel=self.use_cuda_kernel,
            local_files_only=True
        )
    else:
        raise FileNotFoundError(f"Offline mode: BigVGAN model not found at {bigvgan_local} or {bigvgan_name}")
else:
    print(f">> Loading BigVGAN from HuggingFace: {bigvgan_name}")
    self.bigvgan = bigvgan.BigVGAN.from_pretrained(
        bigvgan_name, 
        use_cuda_kernel=self.use_cuda_kernel
    )

## 离线部署准备步骤

在使用 IndexTTS 的离线部署前，需要先下载以下模型：

```bash
# 1. 下载 w2v-bert-2.0
huggingface-cli download facebook/w2v-bert-2.0 \
    --local-dir algorithms/IndexTTS/checkpoints/w2v-bert-2.0

# 2. 下载 semantic_codec
mkdir -p algorithms/IndexTTS/checkpoints/semantic_codec
huggingface-cli download amphion/MaskGCT semantic_codec/model.safetensors \
    --local-dir algorithms/IndexTTS/checkpoints

# 3. 下载 campplus
huggingface-cli download funasr/campplus campplus_cn_common.bin \
    --local-dir algorithms/IndexTTS/checkpoints

# 4. 下载 bigvgan
huggingface-cli download nvidia/bigvgan_v2_22khz_80band_256x \
    --local-dir algorithms/IndexTTS/checkpoints/bigvgan
```
